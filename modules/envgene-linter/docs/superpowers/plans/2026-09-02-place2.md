**Goal:** `envgene-linter check <repo>` prints PLACE-1 first, then PLACE-2 restatement findings.

**Architecture:** Reuse discovery and Effective Set. Add `rules/place2.py`. Engine concatenates PLACE-1 then PLACE-2 findings. Report groups by rule and always prints both headers.

**Tech Stack:** Python 3.12+, pytest, existing `envgene_linter` package.

**Spec (English):** `docs/superpowers/specs/2026-09-02-place2-design.md`  
**Spec (Russian):** `docs/superpowers/specs/ru/2026-09-02-place2-design.md`

## Global Constraints

- Do not change discovery or Effective Set unless a bug blocks PLACE-2.
- Layers: repository, cluster, environment. Only paramsets named in `env_definition`.
- PLACE-2 is warning (SHOULD). Exit codes unchanged (0 on warnings, 2 if no `environments/`).
- Always print both `PLACE-1` and `PLACE-2` headers, even when a rule is silent.
- Cluster restatement: one finding, not one per env.
- A key must not get both PLACE-1 and PLACE-2 (PLACE-1 already skips restatements).
- Algorithm docs: `docs/algorithms/place2.md` and `docs/algorithms/ru/place2.md`.
- Prefer `.venv/bin/python -m pytest`.
- Do not invent `--rules`, JSON, baseline, exceptions, or template/passport.

# src/envgene_linter/rules/place2.py
"""PLACE-2: override only the delta (SHOULD). See docs/algorithms/place2.md."""

from __future__ import annotations

from ..effective import EnvEffective
from ..model import Finding, Layer, RepoIndex, Severity


def check(
    index: RepoIndex,
    full: dict[str, EnvEffective],
    lower: dict[str, EnvEffective],
    site: dict[str, EnvEffective],
) -> list[Finding]:
    findings: list[Finding] = []
    seen: set[tuple[str, str, str]] = set()
    for env in index.environments:
        higher = full.get(env.full_name)
        below = lower.get(env.full_name)
        repo_only = site.get(env.full_name)
        if higher is None or below is None or repo_only is None:
            continue
        _collect(findings, seen, env.full_name, higher, below, Layer.ENVIRONMENT)
        _collect(findings, seen, env.full_name, below, repo_only, Layer.CLUSTER)
    return findings


def _collect(
    findings: list[Finding],
    seen: set[tuple[str, str, str]],
    env_name: str,
    higher: EnvEffective,
    below_map: EnvEffective,
    layer: Layer,
) -> None:
    for scope, mapping in higher.scopes.items():
        for path, leaf in mapping.items():
            if leaf.provenance.layer is not layer:
                continue
            below = below_map.get(scope, path)
            if below is None or below.value != leaf.value:
                continue
            owner = env_name if layer is Layer.ENVIRONMENT else (leaf.provenance.file.cluster or "repository")
            scope_label = f"{owner} {scope}"
            marker = (str(leaf.provenance.file.path), leaf.key, scope_label)
            if marker in seen:
                continue
            seen.add(marker)
            findings.append(
                Finding(
                    rule="PLACE-2",
                    severity=Severity.WARNING,
                    path=leaf.provenance.file.path,
                    line=leaf.provenance.position[0],
                    key=leaf.key,
                    scope=scope_label,
                    message=(
                        f"Key {leaf.key} at the {layer.value} layer repeats the value already "
                        f"supplied by the {below.provenance.layer.value} layer"
                    ),
                    hint=(
                        f"Remove {leaf.key} here, or change the value if a genuine difference "
                        f"was intended"
                    ),
                    related=(f"value {leaf.value!r} already comes from {below.provenance.describe()}",),
                )
            )
```

# src/envgene_linter/report.py
from __future__ import annotations

from .model import Finding

RULE_ORDER = ("PLACE-1", "PLACE-2")


def _block(rule: str, findings: list[Finding]) -> str:
    if not findings:
        return f"{rule}\nNo findings"
    chunks = [
        f"{finding.path}:{finding.line}\n{finding.severity.value}\n{finding.message}\n{finding.hint}"
        for finding in findings
    ]
    return f"{rule}\n" + "\n\n".join(chunks)


def render(findings: list[Finding]) -> str:
    by_rule = {rule: [item for item in findings if item.rule == rule] for rule in RULE_ORDER}
    return "\n\n".join(_block(rule, by_rule[rule]) for rule in RULE_ORDER) + "\n"
```
