> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `envgene-linter check <repo>` prints PLACE-1, then PLACE-2, then PLACE-3 (passport file on the cluster; contract keys at cluster unless an env overrides).

**Architecture:** Extend discovery with passport files and `inventory.cloudPassport`. New `passport.py` resolves one env and builds catalogs (EnvGene table ∪ flattened passport keys). `place3.py` runs three checks. PLACE-1 skips catalog keys. Effective Set merge does not change.

**Tech Stack:** Python 3.12+, pytest, existing `envgene_linter` package.

**Spec (English):** `docs/superpowers/specs/2026-09-04-place3-design.md`  
**Spec (Russian):** `docs/superpowers/specs/ru/2026-09-04-place3-design.md`

## Global Constraints

- Do not merge the passport into Effective Set projections.
- Layers stay repository, cluster, environment. Bound paramsets only for Effective Set.
- PLACE-3 is warning (SHOULD). Exit codes unchanged (0 on warnings, 2 if no `environments/`).
- Always print `PLACE-1`, `PLACE-2`, `PLACE-3` headers, even when a rule is silent.
- Restatement vs lower ParameterSets is PLACE-2 only. One env or differing env values is silent.
- PLACE-1 does not hoist a catalog key.
- Resolution matches the generator but never fails the check (stderr note via `index.skipped`).
- Algorithm docs: `docs/algorithms/place3.md` and `docs/algorithms/ru/place3.md`.
- Prefer `.venv/bin/python -m pytest`.
- Do not invent `--rules`, JSON, baseline, exceptions, template, or PLACE-4.

    def passport(
        self,
        name: str,
        sections: dict | None = None,
        *,
        cluster: str | None = None,
        env: str | None = None,
        folder: str = "cloud-passport",
    ) -> RepoBuilder:
        if cluster and env:
            directory = self.root / "environments" / cluster / env / "Inventory" / folder
        elif cluster:
            directory = self.root / "environments" / cluster / folder
        else:
            directory = self.root / "environments" / folder
        directory.mkdir(parents=True, exist_ok=True)
        lines = ["version: 1.5"]
        for section, values in (sections or {}).items():
            lines.append(f"{section}:")
            for key, value in values.items():
                rendered = value if isinstance(value, str) else repr(value)
                lines.append(f"  {key}: {rendered}")
        (directory / f"{name}.yml").write_text("\n".join(lines) + "\n", encoding="utf-8")
        return self

PASSPORT_FOLDERS = ("cloud-passport", "cloud-passports")
AUTO_FOLDER = "cloud-passport"


@dataclass(frozen=True)
class PassportResolution:
    file: PassportFile | None
    note: str | None = None


@dataclass
class Catalogs:
    env: dict[str, frozenset[str]]
    cluster: dict[str, frozenset[str]]
    repo: frozenset[str]


def flattened_keys(passport: PassportFile) -> set[str]:
    if passport.loaded is None or not isinstance(passport.loaded.doc, dict):
        return set()
    keys: set[str] = set()
    for section, body in passport.loaded.doc.items():
        if section == "version" or not isinstance(body, dict):
            continue
        keys.update(str(key) for key in body)
    return keys


def _matches_name(path: Path, name: str) -> bool:
    from .discovery import paramset_stem

    return paramset_stem(path) == name


def _search_explicit(index: RepoIndex, env: EnvModel, name: str) -> list[PassportFile]:
    environments_dir = index.root / "environments"
    levels = [
        env.path / "Inventory",
        env.path.parent,
        environments_dir,
    ]
    found: list[PassportFile] = []
    seen: set[Path] = set()
    for level in levels:
        for folder in PASSPORT_FOLDERS:
            root = level / folder
            for item in index.passports:
                if item.path in seen:
                    continue
                try:
                    item.path.relative_to(root)
                except ValueError:
                    continue
                if _matches_name(item.path, name):
                    found.append(item)
                    seen.add(item.path)
    return found


def resolve_passport(index: RepoIndex, env: EnvModel) -> PassportResolution:
    if env.cloud_passport:
        matches = _search_explicit(index, env, env.cloud_passport)
        if len(matches) == 1:
            return PassportResolution(file=matches[0])
        return PassportResolution(
            file=None,
            note=(
                f"{env.path / 'Inventory' / 'env_definition.yml'}: "
                f"cloudPassport {env.cloud_passport!r} matched {len(matches)} file(s); skipped"
            ),
        )
    cluster_dir = env.path.parent
    for stem in (env.cluster, "passport"):
        matches = [
            item
            for item in index.passports
            if item.cluster == env.cluster
            and item.env is None
            and item.path.parent.name == AUTO_FOLDER
            and _matches_name(item.path, stem)
        ]
        if len(matches) == 1:
            return PassportResolution(file=matches[0])
        if len(matches) > 1:
            return PassportResolution(
                file=None,
                note=f"{cluster_dir}: auto passport {stem!r} matched {len(matches)} file(s); skipped",
            )
    return PassportResolution(file=None)


def build_catalogs(index: RepoIndex) -> Catalogs:
    env_cats: dict[str, frozenset[str]] = {}
    for env in index.environments:
        resolved = resolve_passport(index, env)
        if resolved.note:
            index.skipped.append(resolved.note)
        extra = flattened_keys(resolved.file) if resolved.file else set()
        env_cats[env.full_name] = TABLE | extra
    cluster_cats: dict[str, frozenset[str]] = {}
    for cluster in index.clusters.values():
        file_keys: set[str] = set()
        for item in index.passports:
            if item.cluster == cluster.name and item.env is None:
                file_keys |= flattened_keys(item)
        union = set(TABLE)
        for env in cluster.environments:
            union |= env_cats.get(env.full_name, TABLE)
        cluster_cats[cluster.name] = frozenset(union | file_keys)
    repo = frozenset(set(TABLE).union(*cluster_cats.values())) if cluster_cats else TABLE
    return Catalogs(env=env_cats, cluster=cluster_cats, repo=repo)
```

# src/envgene_linter/rules/place3.py
"""PLACE-3: place by the system tier (SHOULD). See docs/algorithms/place3.md."""

from __future__ import annotations

from collections import defaultdict

from ..effective import EffectiveLeaf, EnvEffective
from ..model import Finding, Layer, RepoIndex, Scope, Severity
from ..passport import Catalogs

MIN_AGREEMENT = 2


def top_key(leaf: EffectiveLeaf) -> str:
    return str(leaf.path[0]) if leaf.path else leaf.key


def check(
    index: RepoIndex,
    full: dict[str, EnvEffective],
    lower: dict[str, EnvEffective],
    site: dict[str, EnvEffective],
    catalogs: Catalogs,
) -> list[Finding]:
    return [
        *_misplaced_files(index),
        *_repository_keys(index, site, catalogs),
        *_env_agreement(index, full, lower, catalogs),
    ]


def _misplaced_files(index: RepoIndex) -> list[Finding]:
    findings: list[Finding] = []
    for item in index.passports:
        if item.on_cluster or item.is_jinja or item.error:
            continue
        cluster = item.cluster or "<cluster>"
        findings.append(
            Finding(
                rule="PLACE-3",
                severity=Severity.WARNING,
                path=item.path,
                line=1,
                key=item.stem,
                scope=item.cluster or "repository",
                message=f"Passport {item.stem} is not at the cluster layer",
                hint=f"Move it to environments/{cluster}/cloud-passport/",
            )
        )
    return findings


def _repository_keys(
    index: RepoIndex, site: dict[str, EnvEffective], catalogs: Catalogs
) -> list[Finding]:
    findings: list[Finding] = []
    seen: set[tuple[str, str]] = set()
    for env in index.environments:
        projection = site.get(env.full_name)
        if projection is None:
            continue
        for scope, mapping in projection.scopes.items():
            for leaf in mapping.values():
                if leaf.provenance.layer is not Layer.REPOSITORY:
                    continue
                key = top_key(leaf)
                if key not in catalogs.repo:
                    continue
                marker = (str(leaf.provenance.file.path), key)
                if marker in seen:
                    continue
                seen.add(marker)
                findings.append(
                    Finding(
                        rule="PLACE-3",
                        severity=Severity.WARNING,
                        path=leaf.provenance.file.path,
                        line=leaf.provenance.position[0],
                        key=key,
                        scope=f"repository {scope}",
                        message=(
                            f"Key {key} belongs to the Cloud Passport contract but is authored "
                            f"at the repository layer"
                        ),
                        hint=f"Move {key} to a cluster-layer paramset",
                    )
                )
    return findings


def _env_agreement(
    index: RepoIndex,
    full: dict[str, EnvEffective],
    lower: dict[str, EnvEffective],
    catalogs: Catalogs,
) -> list[Finding]:
    findings: list[Finding] = []
    seen: set[tuple[str, str]] = set()
    for cluster in index.clusters.values():
        groups: list[tuple[str, dict[Scope, dict[tuple, EffectiveLeaf]]]] = []
        for env in sorted(cluster.environments, key=lambda item: item.name):
            higher = full.get(env.full_name)
            below = lower.get(env.full_name)
            if higher is None or below is None:
                continue
            catalog = catalogs.env.get(env.full_name, frozenset())
            contribution: dict[Scope, dict[tuple, EffectiveLeaf]] = defaultdict(dict)
            for scope, mapping in higher.scopes.items():
                for path, leaf in mapping.items():
                    if leaf.provenance.layer is not Layer.ENVIRONMENT:
                        continue
                    if top_key(leaf) not in catalog:
                        continue
                    prior = below.get(scope, path)
                    if prior is not None and prior.value == leaf.value:
                        continue
                    contribution[scope][path] = leaf
            groups.append((env.full_name, dict(contribution)))
        by_scope: dict[Scope, list[tuple[str, dict[tuple, EffectiveLeaf]]]] = defaultdict(list)
        for name, contribution in groups:
            for scope, mapping in contribution.items():
                by_scope[scope].append((name, mapping))
        for scope in sorted(by_scope, key=str):
            participants = by_scope[scope]
            if len(participants) < MIN_AGREEMENT:
                continue
            common = set(participants[0][1])
            for _, mapping in participants[1:]:
                common &= set(mapping)
            for path in sorted(common):
                owners = [(name, mapping[path]) for name, mapping in participants]
                first = owners[0][1].value
                if any(owned.value != first for _, owned in owners[1:]):
                    continue
                leaf = owners[0][1]
                key = top_key(leaf)
                marker = (str(leaf.provenance.file.path), key)
                if marker in seen:
                    continue
                seen.add(marker)
                findings.append(
                    Finding(
                        rule="PLACE-3",
                        severity=Severity.WARNING,
                        path=leaf.provenance.file.path,
                        line=leaf.provenance.position[0],
                        key=key,
                        scope=f"{cluster.name} {scope}",
                        message=(
                            f"Key {key} belongs to the Cloud Passport contract and is repeated "
                            f"in environments of cluster {cluster.name}"
                        ),
                        hint=(
                            f"Move {key} to the cluster layer; keep it at an environment only "
                            f"when the value is genuinely different"
                        ),
                        related=tuple(f"{name}: {owned.provenance.describe()}" for name, owned in owners),
                    )
                )
    return findings
```
