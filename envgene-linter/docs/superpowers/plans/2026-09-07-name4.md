**Goal:** Report NAME-4 when a bound ParameterSet stem is not `<subject>-<category>` or bakes cluster/env/ticket/release into the name; TYPE Information, ACTION Review.

**Architecture:** `rules/name4.py` reads `env.bindings` and paramset files already on `RepoIndex`. No discovery change. No `compute`. Catalog, `RULE_ORDER`, and engine grow by one rule.

**Tech Stack:** Python 3.12+, pytest, existing `RepoIndex` / `RepoBuilder`.

**Spec (Russian):** `docs/superpowers/specs/ru/2026-09-07-name4-design.md`

## Global Constraints

- Do not rename files or edit `env_definition`. No autofix.
- Do not call `compute` / Effective Set for NAME-4.
- Do not check YAML keys or the `name` field.
- Bound stems only (listed in at least one `envSpecific*`). Unbound files are silent.
- Filename tokens: `deploy` ← deploy array; `pipeline` ← e2e array; `technical` ← technical array.
- Last token is the category; at least two `-`-tokens (subject required); nothing after the category.
- One finding per failing stem. Scope/ticket message wins if both apply.
- Information / Review. Severity `information`.
- Hint: `Review whether this stem can be <subject>-<category>. Do not rename it if env_definition or other logic still depends on the current spelling.`
- Catalog description: `Bound ParameterSet stem is <subject>-<category>`.
- `RULE_ORDER = ("PLACE-1", "PLACE-2", "PLACE-3", "NAME-1", "NAME-2", "NAME-3", "NAME-4")`.
- A fixture that binds `env-params` is not an empty NAME-4 block.
- Prefer `.venv/bin/python -m pytest` (pyproject already has `pythonpath = ["src"]`).
- Work in an isolated git worktree (not on `master`). Include the NAME-4 specs on the branch if still uncommitted.

```python
"""NAME-4: bound ParameterSet stem is <subject>-<category>."""

from __future__ import annotations

import re
from collections import defaultdict
from pathlib import Path

from ..model import Category, Finding, Location, ParamsetFile, RepoIndex, Severity
from ..rulemeta import RULES

_TICKET = re.compile(r"(?:^|-)(?:ticket|jira|issue)-\d+", re.I)
_RELEASE = re.compile(r"(?:^|-)(?:r\d+(?:-\d+)?|20\d{2}[.-]\d{1,2})")
_TOKEN = {
    Category.DEPLOY: "deploy",
    Category.E2E: "pipeline",
    Category.TECHNICAL: "technical",
}
_HINT = (
    "Review whether this stem can be <subject>-<category>. "
    "Do not rename it if env_definition or other logic still depends on the current spelling."
)


def check(index: RepoIndex) -> list[Finding]:
    files_by_stem: dict[str, list[ParamsetFile]] = defaultdict(list)
    for file in _paramsets(index):
        files_by_stem[file.stem].append(file)
    clusters = set(index.clusters)
    envs = {env.name for env in index.environments}
    findings: list[Finding] = []
    for stem, categories in sorted(_bound(index).items()):
        files = files_by_stem.get(stem)
        if not files:
            continue
        primary = sorted(files, key=lambda item: item.path.as_posix())[0]
        scope = ", ".join(sorted(_TOKEN[item] for item in categories))
        if _bakes(stem, clusters | envs):
            findings.append(_finding(stem, scope, primary.path, _scope_message(stem)))
            continue
        required = {_TOKEN[item] for item in categories}
        parts = stem.split("-")
        if len(parts) < 2 or required != {parts[-1]}:
            token = "/".join(sorted(required))
            findings.append(
                _finding(stem, scope, primary.path, _tail_message(stem, token, scope))
            )
    findings.sort(key=lambda item: (item.path.as_posix(), item.key, item.line))
    return findings


def _bound(index: RepoIndex) -> dict[str, set[Category]]:
    found: dict[str, set[Category]] = defaultdict(set)
    for env in index.environments:
        for category in Category:
            for names in env.bound_targets(category).values():
                for stem in names:
                    found[stem].add(category)
    return found


def _bakes(stem: str, names: set[str]) -> bool:
    if _TICKET.search(stem) or _RELEASE.search(stem):
        return True
    lowered = stem.lower()
    for item in names:
        if not item:
            continue
        needle = item.lower()
        if needle == lowered or f"-{needle}-" in f"-{lowered}-":
            return True
        if lowered.startswith(f"{needle}-") or lowered.endswith(f"-{needle}"):
            return True
    return False


def _scope_message(stem: str) -> str:
    return (
        f"ParameterSet {stem!r} bakes a cluster, environment, ticket or release into the name."
    )


def _tail_message(stem: str, token: str, scope: str) -> str:
    return (
        f"ParameterSet {stem!r} must end with -{token} to match its env_definition binding ({scope})."
    )


def _finding(stem: str, scope: str, path: Path, message: str) -> Finding:
    meta = RULES["NAME-4"]
    return Finding(
        rule="NAME-4",
        severity=Severity.INFORMATION,
        path=path,
        line=1,
        column=1,
        issue_type=meta.default_issue_type,
        action=meta.default_action,
        key=stem,
        scope=scope,
        message=message,
        hint=_HINT,
        related=(),
        locations=(Location(path, 1, 1),),
    )


def _paramsets(index: RepoIndex) -> list[ParamsetFile]:
    out = list(index.site_paramsets)
    for cluster in index.clusters.values():
        out.extend(cluster.paramsets)
        for env in cluster.environments:
            out.extend(env.paramsets)
    return out
```
